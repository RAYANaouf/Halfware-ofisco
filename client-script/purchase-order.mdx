## Coli Commande Achat

<Note>
  Script ID: **Coli Commande Achat** — Doctype: **Purchase Order** — Apply To: **Form**
</Note>

This script is a custom ERPNext **Purchase Order Item** script for handling packaging/collisage quantities.

### In simple terms

It lets you enter:

- **Number of packages/boxes** → `custom_nbr_coli`
- **Quantity per package** → `custom_qte_colissage`

And automatically calculates the normal ERPNext `Qty` field.

The main formula is:

```text
Qty = Number of packages × Quantity per package
```

```javascript
frappe.ui.form.on('Purchase Order Item', {
	 item_code: function(frm, cdt, cdn) {
        fetch_collisage_from_item(frm, cdt, cdn);
    },
    
	uom: function(frm, cdt, cdn) {
         fetch_collisage_from_item(frm, cdt, cdn);
    },
    
	custom_nbr_coli: function(frm, cdt, cdn) {
        calculate_qty(cdt, cdn);
    },
    custom_qte_colissage: function(frm, cdt, cdn) {
        calculate_qty(cdt, cdn);
    }
});

function calculate_qty(cdt, cdn) {
    let row = locals[cdt][cdn];
    if (row.custom_colissage) {
        frappe.model.set_value(cdt, cdn, 'qty', row.custom_nbr_coli * row.custom_qte_colissage);
    }
}

function get_conversion_factor(frm, cdt, cdn) {
    let row = locals[cdt][cdn];
    if (row.item_code && row.uom && row.custom_colissage) {
        frappe.call({
            method: 'frappe.client.get_list',
            args: {
                doctype: 'UOM Conversion Detail',
                filters: { parent: row.item_code, uom: 'Colissage' },
                fields: ['conversion_factor'],
                limit: 1
            },
            callback: function(r) {
                console.log(r);
                if (r.message && r.message.length > 0) {
                    let factor = r.message[0].conversion_factor;
                    frappe.model.set_value(cdt, cdn, 'custom_qte_colissage', factor);
                    
                    if (row.custom_nbr_coli) {
                        frappe.model.set_value(cdt, cdn, 'qty', row.custom_nbr_coli * factor);
                    }
                }
            }
        });
    }
}

// Fonction qui va lire directement dans la table 'uoms' du document parent Item
function fetch_collisage_from_item(frm, cdt, cdn) {
    let row = locals[cdt][cdn];
    if (row.item_code) {
        frappe.model.with_doc('Item', row.item_code, function() {
            let item_doc = frappe.get_doc('Item', row.item_code);
            let found_factor = 0;

            // On parcourt la table enfant 'uoms' de la fiche article
            if (item_doc.uoms && item_doc.uoms.length > 0) {
                let match = item_doc.uoms.find(u => u.uom === 'Colissage');
                if (match) {
                    found_factor = match.conversion_factor;
                }
            }

            // Si le facteur "Colissage" est trouvé dans la table de l'article
            if (found_factor > 0) {
                frappe.model.set_value(cdt, cdn, 'custom_qte_colissage', found_factor);
                
                // Recalcul immédiat si un nombre de colis était déjà saisi
                if (row.nbre_colis) {
                    frappe.model.set_value(cdt, cdn, 'qty', row.nbre_colis * found_factor);
                }
            }
        });
    }
}
```
